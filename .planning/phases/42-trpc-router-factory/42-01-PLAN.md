Add `createCrudRouter()` factory at end of file. The factory accepts:
- `crud`: CRUD instance with getAllArray, getById, create, update, delete
- `insertSchema`: Zod schema for create
- `updateSchema`: Zod schema for update
- `getById`: Optional custom getById function (for routers that include relations)

Returns a router with 5 standard procedures. Use type parameters to infer correct types from schemas.

```typescript
interface CrudRouterConfig<TInsert, TUpdate> {
    crud: {
        getAllArray: (entityId?: number) => Promise<unknown[]>
        getById: (id: number) => Promise<unknown>
        create: (data: TInsert) => Promise<unknown>
        update: (id: number, data: TUpdate) => Promise<unknown>
        delete: (id: number) => Promise<unknown>
    }
    insertSchema: z.ZodType<TInsert>
    updateSchema: z.ZodType<TUpdate>
    getById?: (id: number) => Promise<unknown>  // Override for custom queries with relations
}

export function createCrudRouter<TInsert, TUpdate>(
    config: CrudRouterConfig<TInsert, TUpdate>
) {
    const { crud, insertSchema, updateSchema, getById } = config

    return createTRPCRouter({
        list: adminProcedure
            .input(z.object({ entityId: z.coerce.number().optional() }).optional())
            .query(async ({ input }) => crud.getAllArray(input?.entityId)),

        byId: adminProcedure
            .input(z.coerce.number())
            .query(async ({ input }) => (getById ?? crud.getById)(input)),

        create: adminProcedure
            .input(insertSchema)
            .mutation(async ({ input }) => crud.create(input)),

        update: adminProcedure
            .input(z.object({ id: z.coerce.number(), data: updateSchema }))
            .mutation(async ({ input }) => crud.update(input.id, input.data)),

        delete: adminProcedure
            .input(z.coerce.number())
            .mutation(async ({ input }) => crud.delete(input)),
    })
}
```
