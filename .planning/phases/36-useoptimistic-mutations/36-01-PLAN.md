**From 26-RESEARCH.md - useOptimistic pattern:**
```tsx
import { useOptimistic } from 'react'

const [optimisticValue, setOptimisticValue] = useOptimistic(
  actualValue,
  (current, update) => /* return new optimistic state */
)

async function handleAction(update) {
  setOptimisticValue(update)  // Instant UI update
  await mutation(update)       // Actual operation
  // On error, optimistic state automatically reverts
}
```
