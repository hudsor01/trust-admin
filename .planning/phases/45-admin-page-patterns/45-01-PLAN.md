<task type="auto">
  <name>Task 2: Create FormField wrapper component</name>
  <files>src/components/form-field.tsx</files>
  <action>
Create a wrapper component that consolidates the Label + Input + error pattern used with TanStack Form:

```typescript
interface FormFieldProps<T> {
  form: FormApi<T>
  name: keyof T & string
  label: string
  required?: boolean
  type?: 'text' | 'number' | 'email' | 'date' | 'textarea'
  placeholder?: string
  // For select fields
  options?: Array<{ value: string; label: string }>
}

export function FormField<T>({ form, name, label, required, type = 'text', placeholder, options }: FormFieldProps<T>) {
  return (
    <form.Field name={name}>
      {(field) => (
        <div className="space-y-2">
          <Label htmlFor={name}>{label}{required && ' *'}</Label>
          {options ? (
            <Select value={field.state.value || ''} onValueChange={field.handleChange}>
              <SelectTrigger id={name} onBlur={field.handleBlur}>
                <SelectValue placeholder={placeholder} />
              </SelectTrigger>
              <SelectContent>
                {options.map(opt => <SelectItem key={opt.value} value={opt.value}>{opt.label}</SelectItem>)}
              </SelectContent>
            </Select>
          ) : type === 'textarea' ? (
            <Textarea id={name} value={field.state.value || ''} onChange={e => field.handleChange(e.target.value)} onBlur={field.handleBlur} placeholder={placeholder} />
          ) : (
            <Input id={name} type={type} value={field.state.value || ''} onChange={e => field.handleChange(type === 'number' ? Number(e.target.value) : e.target.value)} onBlur={field.handleBlur} placeholder={placeholder} />
          )}
          {field.state.meta.errors?.length > 0 && (
            <p className="text-sm text-red-500">{getFieldError(field)}</p>
          )}
        </div>
      )}
    </form.Field>
  )
}
```

This reduces ~15 lines per field to ~1 line:
- Before: `<form.Field name="email">{(field) => (<div className="space-y-2"><Label>...</Label><Input>...</Input>{errors...}</div>)}</form.Field>`
- After: `<FormField form={form} name="email" label="Email" type="email" />`

Note: Keep the component simple. Complex fields (grids, conditional visibility) should still use the raw form.Field pattern.
