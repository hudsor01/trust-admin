Create a Server Action for HEMS request submission:

```typescript
'use server'

import { z } from 'zod'
import { db } from '@/db'
import { hemsRequest } from '@/db/schema'

const schema = z.object({
  beneficiaryId: z.coerce.number(),
  entityId: z.coerce.number(),
  category: z.enum(['HEALTH', 'EDUCATION', 'MAINTENANCE', 'SUPPORT']),
  amountRequested: z.string().refine(v => !isNaN(parseFloat(v)) && parseFloat(v) > 0),
  justification: z.string().min(1),
})

export type HemsFormState = {
  error: string | null
  success: boolean
}

export async function submitHemsRequest(
  prevState: HemsFormState,
  formData: FormData
): Promise<HemsFormState> {
  const raw = {
    beneficiaryId: formData.get('beneficiaryId'),
    entityId: formData.get('entityId'),
    category: formData.get('category'),
    amountRequested: formData.get('amountRequested'),
    justification: formData.get('justification'),
  }

  const parsed = schema.safeParse(raw)
  if (!parsed.success) {
    return { error: 'Please fill in all required fields correctly', success: false }
  }

  try {
    await db.insert(hemsRequest).values({
      beneficiaryId: parsed.data.beneficiaryId,
      entityId: parsed.data.entityId,
      category: parsed.data.category,
      amountRequested: parsed.data.amountRequested,
      justification: parsed.data.justification,
      status: 'PENDING',
      requestDate: new Date(),
    })
    return { error: null, success: true }
  } catch (e) {
    console.error('HEMS request submission failed:', e)
    return { error: 'Failed to submit request. Please try again.', success: false }
  }
}
```

Key points:
- Use `'use server'` directive at top
- Accept `(prevState, formData)` signature for useActionState compatibility
- Return serializable state object (no throwing errors)
- Use Zod for validation with `z.coerce.number()` for form data
- Keep tRPC router for admin operations (approve/deny) - don't remove it

Avoid: Don't throw errors - return error state instead. Don't use tRPC inside Server Action (direct db access is fine).
