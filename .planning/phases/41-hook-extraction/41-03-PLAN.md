Create a simple hook that wraps the common CRUD mutation pattern:

```typescript
'use client'

import { trpc } from '@/lib/trpc'

/**
 * Creates standard CRUD mutations with auto-invalidation.
 *
 * Usage:
 * const mutations = useCrudMutations('contact')
 * await mutations.create.mutateAsync(data)
 */
export function useCrudMutations<
  K extends keyof typeof trpc & string
>(routerKey: K) {
  const utils = trpc.useUtils()

  // Access router dynamically
  const router = trpc[routerKey] as {
    create: { useMutation: typeof trpc.contact.create.useMutation }
    update: { useMutation: typeof trpc.contact.update.useMutation }
    delete: { useMutation: typeof trpc.contact.delete.useMutation }
  }

  const invalidate = () => {
    // @ts-expect-error - dynamic router access
    utils[routerKey].list.invalidate()
  }

  const create = router.create.useMutation({
    onSuccess: invalidate,
  })

  const update = router.update.useMutation({
    onSuccess: invalidate,
  })

  const deleteMutation = router.delete.useMutation({
    onSuccess: invalidate,
  })

  return { create, update, delete: deleteMutation }
}
```

**Note:** This approach uses a string key to access the router dynamically. Type inference for mutation inputs/outputs may be looser than direct usage, but the hook still provides value by centralizing invalidation logic.

**Alternative if type inference is lost:** Create resource-specific hooks:
```typescript
export function useContactMutations() {
  return useCrudMutations('contact')
}
```
