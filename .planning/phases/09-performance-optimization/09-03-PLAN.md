### Task 1: Create Pagination component

Create reusable pagination UI component:

**File**: `src/components/pagination.tsx` (new file)

```typescript
import { ChevronLeft, ChevronRight } from "lucide-react"
import { Button } from "@/components/ui/button"

export interface PaginationProps {
  currentPage: number
  pageSize: number
  totalCount: number
  onPageChange: (page: number) => void
  disabled?: boolean
}

/**
 * Pagination controls with prev/next buttons and page info
 *
 * @param currentPage - Current page number (1-indexed)
 * @param pageSize - Number of items per page
 * @param totalCount - Total number of items
 * @param onPageChange - Callback when page changes
 * @param disabled - Disable all buttons (during loading)
 *
 * @example
 * ```typescript
 * <Pagination
 *   currentPage={page}
 *   pageSize={20}
 *   totalCount={totalCount}
 *   onPageChange={(newPage) => setPage(newPage)}
 *   disabled={loading}
 * />
 * ```
 */
export function Pagination({
  currentPage,
  pageSize,
  totalCount,
  onPageChange,
  disabled = false,
}: PaginationProps) {
  const totalPages = Math.ceil(totalCount / pageSize)
  const startItem = (currentPage - 1) * pageSize + 1
  const endItem = Math.min(currentPage * pageSize, totalCount)

  const canGoPrevious = currentPage > 1
  const canGoNext = currentPage < totalPages

  return (
    <div className="flex items-center justify-between px-2 py-4">
      <div className="text-sm text-muted-foreground">
        Showing {startItem} to {endItem} of {totalCount} results
      </div>
      <div className="flex items-center gap-2">
        <Button
          variant="outline"
          size="sm"
          onClick={() => onPageChange(currentPage - 1)}
          disabled={!canGoPrevious || disabled}
        >
          <ChevronLeft className="h-4 w-4 mr-1" />
          Previous
        </Button>
        <div className="text-sm">
          Page {currentPage} of {totalPages}
        </div>
        <Button
          variant="outline"
          size="sm"
          onClick={() => onPageChange(currentPage + 1)}
          disabled={!canGoNext || disabled}
        >
          Next
          <ChevronRight className="h-4 w-4 ml-1" />
        </Button>
      </div>
    </div>
  )
}
```

**Success criteria**: Component renders with prev/next buttons and page info

### Task 3: Add pagination support to TanStack Query hooks

**Note**: Phase 08-02 migrated from custom `use-query.ts` to TanStack Query v5. We now add pagination to individual resource hooks instead of a central factory.

Add paginated query options and hooks to resource queries files:

**File**: `src/hooks/trust-accounting/queries.ts` (example - repeat pattern for other resources)

```typescript
// Add pagination types at top of file
export interface PaginatedResult<T> {
  data: T[]
  totalCount?: number
  limit?: number
  offset?: number
  hasMore?: boolean
}

export interface PaginationParams {
  page: number
  pageSize: number
}

// Add paginated query options
export const trustAccountingPaginatedQueryOptions = (
  entityId: string | undefined,
  pagination: PaginationParams
) =>
  queryOptions({
    queryKey: [
      ...(entityId ? trustAccountingKeys.byEntity(entityId) : trustAccountingKeys.all),
      'paginated',
      pagination.page,
      pagination.pageSize,
    ],
    queryFn: async () => {
      const params = new URLSearchParams()
      if (entityId) params.append('entityId', entityId)
      params.append('limit', String(pagination.pageSize))
      params.append('offset', String((pagination.page - 1) * pagination.pageSize))
      params.append('includeTotalCount', 'true')

      const url = `/api/trust-accounting?${params.toString()}`
      const res = await fetch(url)
      if (!res.ok) {
        const errorData = await res.json().catch(() => ({ error: { message: `HTTP ${res.status}` } }))
        throw new Error(errorData.error?.message || `Failed to fetch: ${res.status}`)
      }
      const result = await res.json() as PaginatedResult<TrustAccounting>

      // Sort the paginated data (if needed)
      if (result.data) {
        result.data.sort((a, b) => new Date(b.accountingDate).getTime() - new Date(a.accountingDate).getTime())
      }

      return result
    },
    enabled: entityId ? !!entityId : true,
    placeholderData: (prev) => prev, // Keep previous data while fetching
  })

// Add paginated hook
export function useTrustAccountingPaginated(
  entityId: string | undefined,
  pagination: PaginationParams
) {
  return useQuery(trustAccountingPaginatedQueryOptions(entityId, pagination))
}
```

**Important TanStack Query patterns**:
- `queryKey` includes pagination params to cache separate pages
- `placeholderData: (prev) => prev` keeps previous data visible during page changes (better UX)
- Separate hook for pagination (keeps existing non-paginated hook for backward compatibility)
- Query is automatically refetched when pagination params change

**Success criteria**:
- TypeScript compiles
- Paginated hook returns `PaginatedResult<T>` with data and totalCount
- URL includes pagination query params
- Page changes trigger automatic refetch via TanStack Query
- Previous page data remains visible while loading new page

## Notes

**Opt-in pagination**: Pages must explicitly enable pagination. Existing pages continue to work without changes.

**Page size**: Default 20 items per page. Can be customized per hook.

**Performance**: Only fetches one page of data at a time, reducing initial load time and memory usage.

**User experience**:
- Clear page indicators (Page X of Y)
- Disabled buttons during loading
- Item count display (Showing X to Y of Z)

**Future enhancements** (not in this plan):
- Page size selector (10, 20, 50, 100)
- Jump to page input
- Keyboard navigation (arrow keys)
- URL state persistence (page in query params)

## Limitations

**Client-side sorting conflicts**: If DataTable has client-side sorting enabled AND pagination is enabled, sorting only affects the current page (not all data). Consider:
- Disable client-side sorting when pagination enabled
- OR add server-side sorting in future phase
- OR document this limitation

**Filter + pagination interaction**: Changing filters resets to page 1. This is correct behavior but should be documented.
