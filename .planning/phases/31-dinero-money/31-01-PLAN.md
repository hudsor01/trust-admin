## Why dinero.js

Database stores money as strings ("1500.00") for precision. Currently, calculations use parseFloat which can introduce floating point errors in sums and comparisons. dinero.js uses integer-based math (cents) to avoid this.

2. Create `src/lib/money.ts` with utility functions:
   ```typescript
   import { USD } from '@dinero.js/currencies'
   import { dinero, add, toDecimal, type Dinero } from 'dinero.js'

   // Convert DB string ("1500.00") to Dinero object
   export function fromString(value: string | null | undefined): Dinero<number> {
     if (!value) return dinero({ amount: 0, currency: USD })
     const cents = Math.round(parseFloat(value) * 100)
     return dinero({ amount: cents, currency: USD })
   }

   // Sum array of DB strings
   export function sumStrings(values: (string | null | undefined)[]): Dinero<number> {
     return values.map(fromString).reduce(add, dinero({ amount: 0, currency: USD }))
   }

   // Format Dinero for display
   export function formatMoney(d: Dinero<number>): string {
     const decimal = toDecimal(d)
     return new Intl.NumberFormat('en-US', {
       style: 'currency',
       currency: 'USD',
     }).format(parseFloat(decimal))
   }

   // Convert Dinero to display string (for backward compat)
   export function toDisplayString(d: Dinero<number>): string {
     return toDecimal(d)
   }
   ```

   ```typescript

   export function formatCurrency(amount: string | number | null): string {
     if (amount === null || amount === undefined) return '$0.00'
     const value = typeof amount === 'number' ? amount.toString() : amount
     return formatMoney(fromString(value))
   }
   ```
