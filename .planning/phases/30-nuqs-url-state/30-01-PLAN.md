<task type="auto">
  <name>Task 1: Install nuqs and create useEntityFilter hook</name>
  <files>package.json, src/app/layout.tsx, src/hooks/use-entity-filter.ts</files>
  <action>
1. Install nuqs:
   ```bash
   bun add nuqs
   ```

2. Update `src/app/layout.tsx` to wrap children with NuqsAdapter:
   ```tsx
   import { NuqsAdapter } from 'nuqs/adapters/next/app'

   // Inside body, wrap TRPCProvider content:
   <NuqsAdapter>
     <TRPCProvider>
       {children}
       <Toaster ... />
     </TRPCProvider>
   </NuqsAdapter>
   ```

3. Create `src/hooks/use-entity-filter.ts`:
   ```tsx
   'use client'
   import { useQueryState, parseAsString } from 'nuqs'

   export function useEntityFilter() {
     return useQueryState('entity', parseAsString.withDefault(''))
   }
   ```

IMPORTANT: NuqsAdapter must be OUTSIDE TRPCProvider (it's a higher-level wrapper).
  </action>
  <verify>
1. `bun run build` succeeds
2. `grep -r "NuqsAdapter" src/app/layout.tsx` shows adapter in layout
3. File exists: `src/hooks/use-entity-filter.ts`
  </verify>
  <done>nuqs installed, NuqsAdapter wrapping app, useEntityFilter hook created</done>
</task>
