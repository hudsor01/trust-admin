  creditor: z.string().min(1, 'Required'),
  currentBalance: z.string().regex(/^\d+\.?\d*$/, 'Invalid amount'),

const bulkEntrySchema = z.object({

1. Create keyboard navigation handler:

```typescript
// Keyboard navigation within the table
const handleKeyDown = useCallback((
  e: React.KeyboardEvent,
  rowIdx: number,
  colIdx: number
) => {
  const colCount = hasLoanTermFields ? 7 : 5

  if (e.key === 'Tab' && !e.shiftKey && colIdx === colCount - 1) {
    // Tab on last column - move to first column of next row
    e.preventDefault()
    if (rowIdx === fields.length - 1) {
      // Last row - add new row and focus it
      append(createEmptyRow())
      // Focus handled in useEffect after render
    } else {
      focusCell(rowIdx + 1, 0)
    }
  }

  if (e.key === 'Enter' && !e.shiftKey) {
    e.preventDefault()
    if (rowIdx === fields.length - 1) {
      append(createEmptyRow())
    } else {
      focusCell(rowIdx + 1, colIdx)
    }
  }

  if (e.key === 'ArrowDown') {
    e.preventDefault()
    focusCell(rowIdx + 1, colIdx)
  }

  if (e.key === 'ArrowUp') {
    e.preventDefault()
    focusCell(rowIdx - 1, colIdx)
  }
}, [fields.length, hasLoanTermFields, append])

const focusCell = (row: number, col: number) => {
  if (row < 0 || row >= fields.length) return
  const input = tableRef.current?.querySelector(
    `[data-row="${row}"][data-col="${col}"] input`
  ) as HTMLInputElement | null
  input?.focus()
  input?.select()
}
```

3. Add clipboard paste handler:

```typescript
const handlePaste = useCallback((e: React.ClipboardEvent) => {
  const text = e.clipboardData.getData('text/plain')
  if (!text) return

  // Excel/Sheets use tab for columns, newline for rows
  const rows = text
    .split('\n')
    .filter(row => row.trim())
    .map(row => row.split('\t'))

  if (rows.length === 0) return

  e.preventDefault()

  // Map pasted data to liability rows
  // Assumes column order: Type, Creditor, Balance, Rate, Payment
  rows.forEach(([type, creditor, balance, rate, payment]) => {
    const liabilityType = mapPastedType(type) || 'OTHER'
    append({
      liabilityType,
      creditor: creditor?.trim() || '',
      currentBalance: balance?.replace(/[$,]/g, '').trim() || '',
      interestRate: rate?.replace(/%/g, '').trim() || '',
      monthlyPayment: payment?.replace(/[$,]/g, '').trim() || '',
      loanTermMonths: '',
      escrowMonthly: '',
    })
  })

  toast.success(`Pasted ${rows.length} rows`)
}, [append])

// Map common type strings to enum values
function mapPastedType(type: string | undefined): string | null {
  if (!type) return null
  const lower = type.toLowerCase().trim()
  if (lower.includes('mortgage')) return 'MORTGAGE'
  if (lower.includes('loan')) return 'LOAN'
  if (lower.includes('credit') || lower.includes('card')) return 'CREDIT_CARD'
  if (lower.includes('tax')) return 'TAX_OWED'
  return 'OTHER'
}
```

const bulkCreateMutation = trpc.liability.bulkCreate.useMutation({
  onSuccess: (results) => {
    utils.liability.list.invalidate()
    toast.success(`Created ${results.length} liabilities`)
    setBulkMode(false)

const handleBulkSave = async (rows: BulkLiabilityRow[]) => {
  if (!selectedEntity) return

  // Transform rows to full liability payloads
  const liabilities = rows.map(row => ({
    liabilityType: row.liabilityType,
    creditor: row.creditor,
    originalAmount: row.currentBalance, // Use current as original for bulk entry
    currentBalance: row.currentBalance,
    interestRate: row.interestRate || null,
    monthlyPayment: row.monthlyPayment || null,
    loanTermMonths: row.loanTermMonths ? parseInt(row.loanTermMonths, 10) : null,
    escrowMonthly: row.escrowMonthly || null,
    isRevolvingCredit: row.liabilityType === 'CREDIT_CARD',
    status: 'ACTIVE' as const,
  }))

  await bulkCreateMutation.mutateAsync({
    entityId: selectedEntity,
    liabilities,
  })
}
